% Searle's Chinese Room (1980).  A note in Chinese goes in; the person inside
% follows the rule book and sends back a reply.  They do not know Chinese.
\def\cjk#1{\includegraphics[height=0.5cm]{cjk/#1}}

% the room
\draw[ink, line width=2pt, fill=ink!4] (3.2,-1.6) rectangle (9.0,2.4);
\node[font=\large\bfseries, text=ink] at (6.1,2.75) {The room};
\fill[white] (3.05,0.75) rectangle (3.35,1.25);  % slot in
\fill[white] (3.05,-0.55) rectangle (3.35,-0.05); % slot out

% person inside with a rule book
\personicon{6.9}{0.1}{1.1}
\draw[machine, line width=1.2pt, fill=machine!20] (4.4,-0.9) -- (5.3,-0.75) -- (6.2,-0.9) -- (6.2,0.05) -- (5.3,0.2) -- (4.4,0.05) -- cycle;
\draw[machine, line width=1.2pt] (5.3,-0.75) -- (5.3,0.2);
\node[font=\small\bfseries, text=machine, align=center] at (5.3,0.65) {rule book};
\node[font=\small, align=center, text=ink] at (6.9,-1.15) {does not know Chinese};

% outside: someone who knows Chinese
\personicon{-1.9}{0.3}{1.1}
\node[font=\small, align=center, text=person] at (-1.9,-0.85) {knows Chinese};

% note in, reply out
\node[draw=ink!50, fill=white, rounded corners=2pt, inner sep=4pt] (qin) at (1.15,1.45) {\cjk{q1}};
\draw[flow] (-1.0,1.0) -- (3.0,1.0);
\node[font=\footnotesize, text=ink!80] at (1.15,0.72) {a note goes in};
\node[draw=result, fill=white, rounded corners=2pt, inner sep=4pt, line width=1pt] (aout) at (1.15,-1.15) {\cjk{a1}};
\draw[flow, draw=result] (3.0,-0.3) -- (-1.0,-0.3);
\node[font=\footnotesize, text=result] at (1.15,-0.58) {a reply comes out};
\end{tikzpicture}
